  _bindEvents() {
    // Header drag
    const header = document.getElementById("bf-header");
    header.addEventListener("mousedown", (e) => {
      if (e.target.tagName === "BUTTON") return;
      this.isDragging = true;
      const rect = this.panel.getBoundingClientRect();
      this.dragOffset = { x: e.clientX - rect.left, y: e.clientY - rect.top };
      this.panel.style.transition = "none";
    });

    document.addEventListener("mousemove", (e) => {
      if (!this.isDragging) return;
      this.panel.style.left = e.clientX - this.dragOffset.x + "px";
      this.panel.style.top = e.clientY - this.dragOffset.y + "px";
      this.panel.style.right = "auto";
    });

    document.addEventListener("mouseup", () => {
      this.isDragging = false;
      this.panel.style.transition = "";
    });

    // Buttons
    document.getElementById("bf-minimize").onclick = () =>
      this.toggleMinimize();
    document.getElementById("bf-close").onclick = () => this.destroy();
    document.getElementById("bf-auto").onclick = () => this.autoFill();
    document.getElementById("bf-scan").onclick = () => this.scanPosition();
    document.getElementById("bf-open").onclick = () => this.openModal();
    document.getElementById("bf-fill").onclick = () => this.fillForm();
  },

  toggleMinimize() {
    this.isMinimized = !this.isMinimized;
    this.panel.classList.toggle("minimized", this.isMinimized);
    document.getElementById("bf-minimize").textContent = this.isMinimized
      ? "+"
      : "−";
  },

  log(msg) {
    const status = document.getElementById("bf-status");
    if (status) {
      const time = new Date().toLocaleTimeString("en-US", {
        hour12: false,
        hour: "2-digit",
        minute: "2-digit",
        second: "2-digit",
      });
      status.innerHTML = `<div>${time} ${msg}</div>` + status.innerHTML;
      // Keep only last 5 messages
      while (status.children.length > 5) {
        status.removeChild(status.lastChild);
      }
    }
  },

  updateInfo(pos) {
    const info = document.getElementById("bf-info");
    if (!pos) {
      info.classList.remove("has-data");
      return;
    }
    info.classList.add("has-data");
    const typeClass = pos.type === "LONG" ? "long" : "short";
    info.innerHTML = `
      <div><span class="label">Position:</span> <span class="value ${typeClass}">${
      pos.type
    }</span> <span class="value">${pos.symbol}</span></div>
      <div><span class="label">Entry:</span> <span class="value">${pos.entry.toFixed(
        2
      )}</span> <span class="label">Type:</span> <span class="value">${
      pos.orderType
    }</span></div>
    `;
  },

  scanPosition() {
    this.log("Scanning...");
    this.position = PositionTool.getSelected();
    if (this.position) {
      this.updateInfo(this.position);
      this.log(`${this.position.type} ${this.position.symbol}`);
    } else {
      this.updateInfo(null);
      this.log("No position found");
    }
  },

  // Force rescan with fresh context (clears all caches)
  rescan() {
    this.log("Force rescanning...");
    PositionTool.clearCache();
    ModalOpener.clearCache();
    this.scanPosition();
  },

  openModal() {
    if (!this.position) {
      this.log("Scan first");
      return;
    }
    const success = ModalOpener.openModal(
      this.position.entry,
      this.position.symbol
    );
    if (success === false && ModalOpener._killed) {
      this.log("Modal opener in cooldown");
    } else {
      this.log(success ? "Modal opened" : "Failed");
    }
  },

  async fillForm() {
    if (!this.position) {
      this.log("Scan first");
      return;
    }

    // Get current price from API
    const currentPrice = PositionTool.getCurrentPrice();
    if (!currentPrice) {
      this.log("Could not get current price");
      return;
    }

    const pos = this.position;
    const isLong = pos.type === "LONG";

    // Calculate order type based on entry vs current price
    const orderType = PositionTool.getOrderType(
      pos.entry,
      isLong,
      currentPrice
    );

    this.log(
      `Filling (${orderType})... entry=${pos.entry.toFixed(
        2
      )} current=${currentPrice.toFixed(2)}`
    );

    const result = await OrderFiller.fill({
      symbol: pos.symbol,
      side: isLong ? "BUY" : "SELL",
      orderType: orderType,
      qty: pos.qty,
      price: pos.entry,
      stopPrice: pos.stopPrice,
      targetPrice: pos.targetPrice,
    });

    if (!result.success) {
      this.log(`Fill failed: ${result.error}`);
      return;
    }

    this.log(`Filled (${orderType}) [${result.attempts} attempt(s)]`);

    // Check protection status after fill
    setTimeout(() => this.checkProtectionStatus(), 300);
  },

  // Check and warn about protection order status
  checkProtectionStatus() {
    const status = OrderFiller.checkProtectionStatus();
    if (!status.enabled) {
      this.log("Protection orders NOT enabled!");
    } else if (!status.slChecked && !status.tpChecked) {
      this.log("SL/TP not checked!");
    } else {
      // Specific warnings
      if (!status.slChecked) {
        this.log("Stop Loss not checked");
      }
      if (!status.tpChecked) {
        this.log("Take Profit not checked");
      }
    }
  },

  async autoFill() {
    this.log("Auto-filling...");
    this.scanPosition();
    if (!this.position) return;

    await this._sleep(200);
    this.openModal();
    await this._sleep(500);
    await this.fillForm();
    this.log("Complete!");
  },

  _sleep(ms) {
    return new Promise((r) => setTimeout(r, ms));
  },

  destroy() {
    if (this.panel) {
      this.panel.remove();
      this.panel = null;
    }
  },
};
